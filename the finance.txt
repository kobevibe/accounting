    (function initMatrixCalculator() {
      const revSlider = document.getElementById('revenue-slider');
      const exitSlider = document.getElementById('exit-slider');
      
      const revVal = document.getElementById('revenue-val');
      const exitVal = document.getElementById('exit-val');
      const shieldVal = document.getElementById('shield-val');
      const rndVal = document.getElementById('rnd-val');
      const alphaVal = document.getElementById('alpha-val');

      function formatCurrency(val) {
        return '$' + Number(val).toLocaleString();
      }

      function updateCalculation() {
        if (!revSlider || !exitSlider) return;
        const rev = parseFloat(revSlider.value);
        const exit = parseFloat(exitSlider.value);

        revVal.textContent = formatCurrency(rev);
        exitVal.textContent = formatCurrency(exit);

        // Section 1202 typical maximum exclusion capped at $10M or 10x basis
        const calculatedShield = Math.min(exit, 10000000);
        shieldVal.textContent = formatCurrency(calculatedShield);

        // Estimated R&D tax credit based on typical tech enterprise qualifying payroll & expenses
        const estimatedRnd = Math.round(rev * 0.032);
        rndVal.textContent = formatCurrency(estimatedRnd);

        // Strategic combined alpha estimate
        const totalAlpha = Math.round((calculatedShield * 0.238) + estimatedRnd + (rev * 0.045));
        alphaVal.textContent = formatCurrency(totalAlpha);
      }

      if (revSlider && exitSlider) {
        revSlider.addEventListener('input', updateCalculation);
        exitSlider.addEventListener('input', updateCalculation);
        updateCalculation();
      }
    })();
